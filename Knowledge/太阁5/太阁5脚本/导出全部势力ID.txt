// ===================================================================
// 用途：全量导出太阁5所有势力（武家/商家/忍者/海贼/流派）的 番号（ID）+ 势力名 + 当主
// 原理（全部取自官方剧本已验证语法，参照 导出全部角色ID.txt 已跑通模式）：
//   容器設定:(大名家,存在,真)    = 遍历所有存在的武家（官方剧本 137 处先例）
//   容器設定:(商家,存在,真)      = 遍历所有商家（官方 8 处先例）
//   容器設定:(流派,存在,真)      = 遍历所有流派（官方 1 处先例）
//   忍者眾/海賊眾 无容器先例 → 改走人物遍历：人物Ａ.所屬忍者眾 / 所屬海賊眾
//   代入ａ:(大名家Ａ.大名家番號)  = 番号属性（database.xml 屬性表 id=0，五类同名）
//   [[<大名家Ａ>]]               = 势力名（tkhack 显示当主名）
//   日誌:[[...]]                  = 输出到 E:\TKHACK\log\ 日志（一行一条）
// 用法：事件管理 → 添加事件 → 加载本文件 → 大地图上移动一格即触发 → 去
//   E:\TKHACK\log\ 抓日志行。
// 输出格式：類型|番号|势力名|当主|本據（| 分隔，一行一个势力）
// ===================================================================

事件:導出全部勢力ID{
    屬性:一次
    發生契機:移動畫面表示後(無效)
    發生條件:{
    }
    執行:{
        自語:[[勢力導出開始，先導出武家……]]
        // ---------- 1. 武家（大名家）：番号 = 当主人物番号 + 70 ----------
        容器設定:(大名家,存在,真)
        代入ｍ:(容器記錄數)
        條件循環:(容器記錄數)!=(0){
            代入大名家Ａ:(無效)
            容器選擇:(大名家Ａ,先頭)
            容器清理:(消去,1)
            代入ａ:(大名家Ａ.大名家番號)
            代入人物Ｂ:(大名家Ａ.當主)
            代入據點Ａ:(大名家Ａ.本城)
            日誌:[[大名家|<ａ>|<大名家Ａ>|當主:<人物Ｂ>|本據:<據點Ａ>]]
        }
        自語:[[武家<ｍ>個已寫入日誌，繼續導出商家……]]
        // ---------- 2. 商家（屋号）：番号 = 商家名索引 + 50 ----------
        容器設定:(商家,存在,真)
        代入ｍ:(容器記錄數)
        條件循環:(容器記錄數)!=(0){
            代入商家Ａ:(無效)
            容器選擇:(商家Ａ,先頭)
            容器清理:(消去,1)
            代入ａ:(商家Ａ.商家番號)
            代入人物Ｂ:(商家Ａ.當主)
            日誌:[[商家|<ａ>|<商家Ａ>|當主:<人物Ｂ>]]
        }
        自語:[[商家<ｍ>個已寫入日誌，繼續導出忍者眾……]]
        // ---------- 3. 忍者眾：无容器 → 人物遍历（去重以番号为键） ----------
        容器設定:(人物,無效,無效)
        條件循環:(容器記錄數)!=(0){
            代入人物Ａ:(無效)
            容器選擇:(人物Ａ,先頭)
            容器清理:(消去,1)
            代入忍者眾Ａ:(人物Ａ.所屬忍者眾)
            代入ａ:(忍者眾Ａ.忍者眾番號)
            日誌:[[忍者眾|<ａ>|<忍者眾Ａ>|所屬人物:<人物Ａ>]]
        }
        自語:[[忍者眾已按人物反查寫入日誌（可能有重复行，按番号去重），繼續導出海賊眾……]]
        // ---------- 4. 海賊眾：同忍者眾，走人物反查 ----------
        容器設定:(人物,無效,無效)
        條件循環:(容器記錄數)!=(0){
            代入人物Ａ:(無效)
            容器選擇:(人物Ａ,先頭)
            容器清理:(消去,1)
            代入海賊眾Ａ:(人物Ａ.所屬海賊眾)
            代入ａ:(海賊眾Ａ.海賊眾番號)
            日誌:[[海賊眾|<ａ>|<海賊眾Ａ>|所屬人物:<人物Ａ>]]
        }
        自語:[[海賊眾已寫入日誌，繼續導出流派……]]
        // ---------- 5. 流派 ----------
        容器設定:(流派,存在,真)
        代入ｍ:(容器記錄數)
        條件循環:(容器記錄數)!=(0){
            代入流派Ａ:(無效)
            容器選擇:(流派Ａ,先頭)
            容器清理:(消去,1)
            代入ａ:(流派Ａ.流派番號)
            代入人物Ｂ:(流派Ａ.宗家)
            日誌:[[流派|<ａ>|<流派Ａ>|宗家:<人物Ｂ>]]
        }
        自語:[[全部勢力已寫入日誌：武家/商家以容器遍历为准，忍者眾/海賊眾為人物反查去重結果。]]
    }
}
}//腳本
